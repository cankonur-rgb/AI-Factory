from spitfire import Pipeline, FeatureStore, ModelRegistry
from spitfire.models import XGBForecaster
from spitfire.monitoring import DriftSentinel, SHAPInsightAgent
import pandas as pd

# 1. Veri kaynaklarını bağla (Feature Store'dan)
def build_pipeline():
    fs = FeatureStore(name="feast_main")
    features = fs.get_features([
        "transaction_amount", "customer_age",
        "product_category", "channel",
        "promo_flag", "prev_purchase_days",
        "region_code", "weekday"
    ])

    # 2. Model tanımı · AutoML tarafından önerilen parametreler
    model = XGBForecaster(
        n_estimators=847,
        max_depth=8,
        learning_rate=0.04,
        subsample=0.82,
        forecast_horizon=7,
        cv_folds=5
    )

    # 3. Monitoring agent'larını otomatik attach et
    pipeline = Pipeline(
        name="demand_forecaster_v3",
        features=features,
        target="demand_qty_next7d",
        model=model,
        monitors=[DriftSentinel(), SHAPInsightAgent()]
    )

    return pipeline

# 4. Deploy ile registry'ye kaydet
if __name__ == "__main__":
    pipeline = build_pipeline()
    pipeline.train(train_window="18M", validation="time_series_cv")
    ModelRegistry.push(pipeline, version="v3.2.1")
